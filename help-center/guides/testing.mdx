## Quick Start

```bash
# Run all tests
pytest

# Run with verbose output
pytest -v

# Run specific file/class/method
pytest tests/test_ticker_symbol_handling.py
pytest tests/test_model_validation.py::TestModelValidation
pytest tests/test_ticker_symbol_handling.py::TickerSymbolHandlingTests::test_normalize_ticker_symbol_preserves_exchange_suffix

# Run by pattern
pytest -k "ticker"

# Run with coverage
pytest --cov=tradingagents --cov=paper_trader --cov=signal_scanner --cov-report=html
```

    ```python
    import unittest
    from my_module import my_function

    class TestMyFeature(unittest.TestCase):

        def setUp(self):
            self.sample_data = {"ticker": "INFY.NS", "price": 1850}

        def tearDown(self):
            pass  # cleanup connections, temp files

        def test_happy_path(self):
            result = my_function(self.sample_data)

        def test_error_handling(self):
            with self.assertRaises(ValueError):
                my_function({"ticker": None, "price": 1850})

        def test_edge_case(self):
            result = my_function({"ticker": "X", "price": 0})
            self.assertIsNotNone(result)

    if __name__ == "__main__":
        unittest.main()
    ```

## Test Categories

<Tabs>
  <Tab title="Unit Tests" icon="box">
    Test individual functions in isolation — no external dependencies. Target: 10–100ms per test.

    ```python
    # tests/test_technical_indicators.py
    from technical import calculate_rsi

    class TestTechnicalIndicators(unittest.TestCase):

        def test_rsi_calculation(self):
            prices = [100, 102, 101, 103, 105, 104, 106, 108, 107, 109]
            rsi = calculate_rsi(prices)
            self.assertGreater(rsi, 0)
            self.assertLess(rsi, 100)

        def test_rsi_empty_data(self):
            with self.assertRaises(ValueError):
                calculate_rsi([])

        def test_rsi_single_value(self):
            rsi = calculate_rsi([100])
            self.assertEqual(rsi, 50)  # neutral
    ```
  </Tab>
  <Tab title="Integration Tests" icon="link">
    Test multiple components together using a test database. Target: 100ms–10s per test.

    ```python
    # tests/test_paper_trader_integration.py
    import tempfile
    from paper_trader import PaperTrader

    class TestPaperTraderIntegration(unittest.TestCase):

        def setUp(self):
            self.db_file = tempfile.NamedTemporaryFile(suffix=".db", delete=False)
            self.pt = PaperTrader(initial_capital=100_000)

        def tearDown(self):
            import os; os.unlink(self.db_file.name)

        def test_enter_and_exit_trade(self):
            entry_id = self.pt.enter_trade(
                symbol="INFY.NS", quantity=10,
                entry_price=1850, stop_loss=1750, target_price=1950
            )
            self.assertIsNotNone(entry_id)
            positions = self.pt.get_open_positions()
            self.assertEqual(len(positions), 1)

            self.pt.exit_trade(entry_id, exit_price=1900, reason="MANUAL")
            positions = self.pt.get_open_positions()
            self.assertEqual(len(positions), 0)

        def test_daily_loss_limit_circuit_breaker(self):
            # Make losing trades until daily_loss > 2%
            for i in range(10):
                try:
                    self.pt.enter_trade("INFY", 1, 1000, 990, 1010)
                    self.pt.exit_trade(..., exit_price=990, reason="LOSS")
                except Exception as e:
                    if "daily loss limit" in str(e):
                        break
            with self.assertRaises(RuntimeError) as ctx:
                self.pt.enter_trade("TCS", 1, 3000, 2900, 3100)
            self.assertIn("daily loss limit", str(ctx.exception))
    ```

    ```python
    # tests/test_e2e_analysis_pipeline.py
    from fastapi.testclient import TestClient
    from api_server import app

    class TestE2EAnalysisPipeline(unittest.TestCase):

        @classmethod
        def setUpClass(cls):
            cls.client = TestClient(app)

        def test_full_analysis_request(self):
            response = self.client.post(
                "/api/analyze/stream",
                json={"ticker": "INFY.NS", "trade_date": "2026-05-03"}
            )
            self.assertEqual(response.status_code, 200)

            lines = response.text.strip().split("\n")
            events = [l for l in lines if l.startswith("data:")]
            self.assertGreater(len(events), 5)

            import json
            final = json.loads(events[-1].split("data: ")[1])
            self.assertIn("decision", final)
            self.assertIn("confidence", final)
    ```

## Async Testing

```python
import asyncio, pytest
from signal_scanner import run_signal_scan

# unittest style
class TestAsyncFunctions(unittest.TestCase):
    def test_signal_scan(self):
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        try:
            result = loop.run_until_complete(
                run_signal_scan(mock_client, mock_trader)
            )
            self.assertIn("scans", result)
        finally:
            loop.close()

# pytest style (requires pip install pytest-asyncio)
@pytest.mark.asyncio
async def test_signal_scan_pytest():
    result = await run_signal_scan(mock_client, mock_trader)
    assert "scans" in result
```

## CI/CD Integration

```yaml
# .github/workflows/tests.yml
name: Tests
on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        python-version: ['3.10', '3.11', '3.12']
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-python@v4
        with:
          python-version: ${{ matrix.python-version }}
      - run: pip install -e . pytest pytest-cov pytest-asyncio
      - run: pytest --cov=tradingagents --cov=paper_trader --cov-report=xml
      - uses: codecov/codecov-action@v3
        with:
          files: ./coverage.xml
```

## Best Practices

**Do:**
- ✅ Test happy path, error cases, and edge cases (boundaries: 0, negative, very large)
- ✅ Mock all external services (Upstox API, LLM providers)
- ✅ Use fixtures for reusable test data and setup
- ✅ Test concurrency — run parallel threads to verify thread safety
- ✅ Verify database state after operations
- ✅ Use `setUp`/`tearDown` to clean up resources

**Don't:**
- ❌ Test implementation details — test behavior, not internal code
- ❌ Use random test data — use deterministic fixtures
- ❌ Run tests against real APIs — use mocks
- ❌ Use hardcoded file paths — use `tempfile`
- ❌ Leave tests failing — fix immediately, don't skip

```bash
# List all tests without running
pytest --collect-only
```
